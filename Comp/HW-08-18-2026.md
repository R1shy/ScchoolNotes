
## Ex 1-5

1. $3x + 7 = -8$
		$3x = -15$
		$x = -5$
2. $2x + 3 = x -1$
		$2x + 4 = x$
		NS
3.  $5x + 3 = 2x + 3$
		$3x + 3 = 3$
		$3x = 3$
		$x = 0$
4. $4x + 3 = x - 8$
	1. {rational only}
		$4x = x - 11$
		$3x = -11$
		$x = \frac{-11}{3}$
	2. {ints}
		NS
5. $x^2 = 16$
	1. {neg only}
		$\sqrt{x^2} = \sqrt{16}$
		$x = 4$
		$4 \gt 0$
		NS
	2. {real only}
		$x = 4$
6. $x^2 = \frac{1}{9}$
	1. {rational only}
		$\sqrt{x} = \sqrt{\frac{1}{9}}$
		$x=0.\overline{333}$
	2. {ints}
		NS
7. $(x+3)(3x-2)=0$
		$x+3=0$
		$x=-3$
		
		$3x-2=0$
		$3x=2$
		$x=\frac{2}{3}$
8. $(2x-5)(x+1)=0$
		$x+1=0$
		$x=-1$
		
		$2x-5=0$
		$2x=5$
		$x=\frac{5}{2}$
9.	$(x+3)(3x-2)=0$
		$x+3=0$
		$x=-3$
		
		$3x-2=0$
		$3x=2$
		$x=2/3$
10. $(2x+3)(x-5)(x+1)=0$
		$2x+3=0$
		$2x=-3$
		$x=\frac{-3}{2}$
		
		$x-5=0$
		$x=5$
		
		$x+1=0$
		$x=-1$
11. $x(2x-1)(x+4)=0$
		$x=0$
		
		$2x-1=0$
		$2x=1$
		$x=\frac{1}{2}$
		
		$x+4=0$
		$x=4$
12. $|x|=7$
		$x=7$ or $x=-7$
